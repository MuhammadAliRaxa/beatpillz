1. Overview & Authentication Architecture

All mobile endpoints are grouped under the prefix /api/v1/*. Every request and response uses pure JSON payloads with standardized structure.

Base URLs:
• Production: https://beatpillz.com/api/v1
• Local Development: http://127.0.0.1:8000/api/v1

Standard Response Format

All successful responses return HTTP status 200 or 201:

{
  "success": true,
  "message": "Optional descriptive message",
  "data": { ... }
}

All validation errors return HTTP status 422:

{
  "success": false,
  "message": "The email has already been taken.",
  "errors": {
    "email": ["The email has already been taken."]
  }
}

2. Authentication & Social Sign-In

POST /auth/register

Register new user with name, username, email, password, and optional device name.

// Request
{
  "firstname": "John",
  "lastname": "Doe",
  "username": "johndoe",
  "email": "johndoe@example.com",
  "password": "Password123",
  "password_confirmation": "Password123",
  "device_name": "Flutter iOS"
}

// Response (201 Created)
{
  "success": true,
  "message": "Registration successful.",
  "access_token": "1|qWeRtY123456...",
  "token_type": "Bearer",
  "user": {
    "id": 1,
    "firstname": "John",
    "lastname": "Doe",
    "fullname": "John Doe",
    "username": "johndoe",
    "email": "johndoe@example.com",
    "avatar": null,
    "is_author": false,
    "balance": 0.00
  }
}
POST /auth/login

Authenticate with Email or Username.

// Request
{
  "login": "johndoe@example.com", // Or "johndoe"
  "password": "Password123",
  "device_name": "iPhone 15 Pro"
}

// Response (200 OK)
{
  "success": true,
  "message": "Login successful.",
  "access_token": "2|aBcDeF654321...",
  "token_type": "Bearer",
  "user": {
    "id": 1,
    "username": "johndoe",
    "email": "johndoe@example.com",
    "balance": 25.50
  }
}
POST /auth/social-login

Native Google, Apple, or Facebook mobile sign-in.

// Request
{
  "provider": "google", // "google" | "apple" | "facebook"
  "provider_id": "10982347209384",
  "email": "johndoe@gmail.com",
  "firstname": "John",
  "lastname": "Doe",
  "avatar": "https://lh3.googleusercontent.com/...",
  "device_name": "Pixel 8"
}

// Response (200 OK)
{
  "success": true,
  "message": "Social login successful.",
  "access_token": "3|zXcVbN987654...",
  "token_type": "Bearer",
  "user": { "id": 1, "username": "johndoe", "email": "johndoe@gmail.com" }
}
POST /auth/forgot-password

Dispatches password reset instructions to user's registered email address.

// Request
{ "email": "johndoe@example.com" }

// Response (200 OK)
{ "success": true, "message": "Password reset instructions have been sent to your email address." }
POST /auth/logout Token Required

Revokes the active mobile session Bearer access token.

// Response (200 OK)
{ "success": true, "message": "Logged out successfully." }

3. User Profile, KYC & Creator Setup

GET /user/profile Token Required

Returns authenticated user profile, balances, KYC status, author status, and badges.

PUT /user/profile Token Required

Update personal info, biography, and social media handles.

// Request Body
{
  "firstname": "John",
  "lastname": "Doe",
  "profile_heading": "Producer & Audio Engineer",
  "profile_description": "Crafting hard trap and afrobeat sounds for 5+ years.",
  "social_links": {
    "instagram": "https://instagram.com/johndoe",
    "youtube": "https://youtube.com/@johndoe"
  }
}
POST /user/avatar Token Required

Upload avatar image file using multipart/form-data (key: avatar, allowed: jpg, png, webp, max: 4MB).

PUT /user/password Token Required

Change password (requires current_password, password, password_confirmation).

GET /user/kyc Token Required

Returns KYC verification status (is_verified: true|false) and submission logs.

POST /user/become-author Token Required

Upgrades standard buyer account to Author/Producer status with default level badge.

PUT /user/withdrawal-account Token Required

Configure author payout account (e.g. withdrawal_method_id: 1, withdrawal_account: "paypal@example.com").

12. Complete Flutter / Dart Service Boilerplate

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

class BeatPillzApi {
  static const String baseUrl = 'https://beatpillz.com/api/v1';
  static const _storage = FlutterSecureStorage();

  // Authentication Headers
  static Future<Map<String, String>> getHeaders() async {
    final token = await _storage.read(key: 'auth_token');
    return {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      if (token != null) 'Authorization': 'Bearer $token',
    };
  }

  // 1. Email/Username Login
  static Future<Map<String, dynamic>> login(String login, String password) async {
    final res = await http.post(
      Uri.parse('$baseUrl/auth/login'),
      headers: {'Content-Type': 'application/json', 'Accept': 'application/json'},
      body: jsonEncode({'login': login, 'password': password, 'device_name': 'Flutter App'}),
    );
    final data = jsonDecode(res.body);
    if (res.statusCode == 200 && data['success'] == true) {
      await _storage.write(key: 'auth_token', value: data['access_token']);
    }
    return data;
  }

  // 2. Google / Apple Social Sign-In
  static Future<Map<String, dynamic>> socialLogin({
    required String provider,
    required String providerId,
    required String email,
    String? firstname,
    String? lastname,
    String? avatar,
  }) async {
    final res = await http.post(
      Uri.parse('$baseUrl/auth/social-login'),
      headers: {'Content-Type': 'application/json', 'Accept': 'application/json'},
      body: jsonEncode({
        'provider': provider,
        'provider_id': providerId,
        'email': email,
        'firstname': firstname,
        'lastname': lastname,
        'avatar': avatar,
        'device_name': 'Flutter App',
      }),
    );
    final data = jsonDecode(res.body);
    if (res.statusCode == 200 && data['success'] == true) {
      await _storage.write(key: 'auth_token', value: data['access_token']);
    }
    return data;
  }

  // 3. Home Feed
  static Future<Map<String, dynamic>> getHome() async {
    final headers = await getHeaders();
    final res = await http.get(Uri.parse('$baseUrl/home'), headers: headers);
    return jsonDecode(res.body);
  }

  // 4. Search Beats Catalog
  static Future<Map<String, dynamic>> getBeats({
    String? query,
    String? category,
    String sort = 'latest',
    int page = 1,
  }) async {
    final headers = await getHeaders();
    final uri = Uri.parse('$baseUrl/items').replace(queryParameters: {
      if (query != null) 'q': query,
      if (category != null) 'category': category,
      'sort': sort,
      'page': '$page',
    });
    final res = await http.get(uri, headers: headers);
    return jsonDecode(res.body);
  }

  // 5. 1-Click Wallet Checkout
  static Future<Map<String, dynamic>> payWithWallet(int transactionId) async {
    final headers = await getHeaders();
    final res = await http.post(
      Uri.parse('$baseUrl/checkout/pay-with-balance'),
      headers: headers,
      body: jsonEncode({'transaction_id': transactionId}),
    );
    return jsonDecode(res.body);
  }

  // 6. User Purchased Library
  static Future<Map<String, dynamic>> getUserPurchases() async {
    final headers = await getHeaders();
    final res = await http.get(Uri.parse('$baseUrl/user/purchases'), headers: headers);
    return jsonDecode(res.body);
  }
}
